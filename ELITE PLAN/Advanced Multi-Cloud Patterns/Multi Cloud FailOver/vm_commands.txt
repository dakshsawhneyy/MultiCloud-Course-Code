#!/bin/bash

set -e

# ============================================================
# Multi-Cloud Microservices VM Bootstrap
# Product / Order / User Services
# ============================================================

# ---- Configuration ------------------------------------------

DOCKER_USERNAME="dakshsawhneyy"
IMAGE_TAG="latest"

PRODUCT_IMAGE="${DOCKER_USERNAME}/multi-cloud_over_failure-product-service:${IMAGE_TAG}"
ORDER_IMAGE="${DOCKER_USERNAME}/multi-cloud_over_failure-order-service:${IMAGE_TAG}"
USER_IMAGE="${DOCKER_USERNAME}/multi-cloud_over_failure-user-service:${IMAGE_TAG}"

# ============================================================
# 1. Update system
# ============================================================

echo "========================================="
echo "Updating system..."
echo "========================================="

sudo apt update

# ============================================================
# 2. Install Docker
# ============================================================

echo "========================================="
echo "Installing Docker..."
echo "========================================="

sudo apt install -y docker.io

sudo systemctl enable --now docker

# Allow current user to run Docker without sudo
sudo usermod -aG docker "$USER" || true

echo "Docker installed:"
sudo docker --version

# ============================================================
# 3. Pull exactly the same images
# ============================================================

echo "========================================="
echo "Pulling application images..."
echo "========================================="

sudo docker pull "$PRODUCT_IMAGE"
sudo docker pull "$ORDER_IMAGE"
sudo docker pull "$USER_IMAGE"

# ============================================================
# 4. Remove old containers if they exist
# ============================================================

echo "========================================="
echo "Removing old containers..."
echo "========================================="

sudo docker rm -f product-service 2>/dev/null || true
sudo docker rm -f order-service 2>/dev/null || true
sudo docker rm -f user-service 2>/dev/null || true

# ============================================================
# 5. Start Product Service
# ============================================================

echo "========================================="
echo "Starting Product Service..."
echo "========================================="

sudo docker run -d \
  --restart unless-stopped \
  --name product-service \
  -p 3001:3001 \
  "$PRODUCT_IMAGE"

# ============================================================
# 6. Start Order Service
# ============================================================

echo "========================================="
echo "Starting Order Service..."
echo "========================================="

sudo docker run -d \
  --restart unless-stopped \
  --name order-service \
  -p 3002:3002 \
  "$ORDER_IMAGE"

# ============================================================
# 7. Start User Service
# ============================================================

echo "========================================="
echo "Starting User Service..."
echo "========================================="

sudo docker run -d \
  --restart unless-stopped \
  --name user-service \
  -p 3003:3003 \
  "$USER_IMAGE"

# ============================================================
# 8. Verify containers
# ============================================================

echo "========================================="
echo "Running containers:"
echo "========================================="

sudo docker ps

# ============================================================
# 9. Test local health endpoints
# ============================================================

echo "========================================="
echo "Testing health endpoints..."
echo "========================================="

sleep 5

echo ""
echo "Product Service:"
curl -f http://localhost:3001/health || echo "Product service health check failed"

echo ""
echo "Order Service:"
curl -f http://localhost:3002/health || echo "Order service health check failed"

echo ""
echo "User Service:"
curl -f http://localhost:3003/health || echo "User service health check failed"

echo ""
echo "========================================="
echo "VM setup completed"
echo "========================================="
